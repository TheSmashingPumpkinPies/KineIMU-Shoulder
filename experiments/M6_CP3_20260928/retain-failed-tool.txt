"""Preserve complete failed batch02 bytes and hashes, never relabel acceptance."""
import json
import shutil
import sys
from hashlib import sha256
from pathlib import Path

EVIDENCE = Path(__file__).resolve().parent
source = Path(sys.argv[1]).resolve()
target = EVIDENCE / "failed-batch02"
assert source.is_dir() and not target.exists()
summary = json.loads((source / "summary.json").read_bytes())
assert summary["cp3_passed"] is False and summary["timed_count"] == 5


def snapshot(root):
    return {p.relative_to(root).as_posix(): sha256(p.read_bytes()).hexdigest()
            for p in sorted(root.rglob("*")) if p.is_file()}


before = snapshot(source)
shutil.copytree(source, target)
assert snapshot(target) == snapshot(source) == before
with (EVIDENCE / "failed-retention.json").open("x", encoding="utf-8", newline="\n") as stream:
    stream.write(json.dumps(dict(original_root=str(source), retained_root=str(target),
                                exact_bytes=True, disposition="FAIL", files=len(before), sha256=before),
                           sort_keys=True, indent=2) + "\n")
print("failed batch retained exactly", len(before))
