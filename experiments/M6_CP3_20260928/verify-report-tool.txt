"""Verify all public table cells against separately derived raw-ns values."""
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
EVIDENCE = Path(__file__).resolve().parent
audit = json.loads((EVIDENCE / "report-values-audit.json").read_bytes())
report = (ROOT / "benchmarks/demo/REPORT.md").read_text(encoding="utf-8")
assert audit["passed"]
for row in audit["attempts"]:
    assert (f"| {row['slot']} | {row['wall_ns']} | {row['wall_s_display']} | {row['rate_display']} | "
            "0 / PASS |") in report
for name, label in (("wall_s", "End-to-end wall_s"), ("node_samples_per_s", "End-to-end node-samples/s")):
    row = audit["displayed_summary"][name]
    assert f"| {label} | {row['median']} | {row['min']} | {row['max']} |" in report
assert audit["source_head"] in report
assert json.loads((EVIDENCE / "original-recompute.json").read_bytes()) == json.loads(
    (EVIDENCE / "retained-recompute.json").read_bytes())
print("all raw-ns cells, six attempt values, two summary rows, source lock and independent copies verified")
