"""Independently derive every displayed performance number from integer ns using rational arithmetic."""
import json
import math
from fractions import Fraction
from hashlib import sha256
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
EVIDENCE = Path(__file__).resolve().parent
BATCH = EVIDENCE / "batch01"
attempts = json.loads((BATCH / "attempts.json").read_bytes())
summary = json.loads((BATCH / "summary.json").read_bytes())
lock = json.loads((BATCH / "lock.json").read_bytes())
env = json.loads((BATCH / "environment-before.json").read_bytes())
assert summary["cp3_passed"] and all(r["disposition"] == "PASS" for r in attempts)
assert [r["slot"] for r in attempts] == ["warmup", "timed01", "timed02", "timed03", "timed04", "timed05"]
N = lock["node_samples"]
timed = attempts[1:]
walls = sorted(Fraction(r["wall_ns"], 10**9) for r in timed)
rates = sorted(Fraction(N * 10**9, r["wall_ns"]) for r in timed)
expected = {"wall_s": dict(median=walls[2], min=walls[0], max=walls[-1]),
            "node_samples_per_s": dict(median=rates[2], min=rates[0], max=rates[-1])}
for name, values in expected.items():
    for stat, rational in values.items():
        actual = summary[name][stat]
        # Independent exact rational vs runner's two floating operations: allow only <=2 float ULPs.
        assert abs(float(rational) - actual) <= 2 * math.ulp(actual)
        assert f"{float(rational):.6f}" == f"{actual:.6f}"
rows = []
for r in attempts:
    assert r["wall_ns"] == r["counter_end_ns"] - r["counter_start_ns"] > 0
    wall = Fraction(r["wall_ns"], 10**9)
    rate = Fraction(N * 10**9, r["wall_ns"])
    assert float(wall) == r["wall_s"]
    assert abs(float(rate) - r["node_samples_per_s"]) <= 2 * math.ulp(r["node_samples_per_s"])
    rows.append(dict(slot=r["slot"], wall_ns=r["wall_ns"], wall_s_display=f"{float(wall):.6f}",
                     rate_display=f"{float(rate):.6f}", rate_exact=[rate.numerator, rate.denominator]))
values = {name: {stat: f"{float(value):.6f}" for stat, value in group.items()} for name, group in expected.items()}
audit = dict(source_head=lock["source_head"], passed=True, warmup_excluded=True,
             method="integer ns -> exact Fraction; independently sorted third/endpoints; float comparison <=2 ULP",
             attempts=rows, displayed_summary=values)
with (EVIDENCE / "report-values-audit.json").open("x", encoding="utf-8", newline="\n") as stream:
    stream.write(json.dumps(audit, sort_keys=True, indent=2) + "\n")
