"""Read-only diagnosis of Windows venv redirector versus actual interpreter PID."""
import json
import os
import subprocess
import sys
from pathlib import Path

command = [sys.executable, "-c", "import os,json; print(json.dumps(dict(pid=os.getpid(),ppid=os.getppid())))"]
p = subprocess.Popen(command, stdout=subprocess.PIPE, text=True)
worker = json.loads(p.communicate()[0])
print(json.dumps(dict(parent_pid=os.getpid(), launch_pid=p.pid, worker=worker,
                      child_parent_is_launch=worker['ppid'] == p.pid,
                      executable=sys.executable, base_executable=sys._base_executable,
                      executable_bytes=Path(sys.executable).stat().st_size,
                      base_bytes=Path(sys._base_executable).stat().st_size)))
