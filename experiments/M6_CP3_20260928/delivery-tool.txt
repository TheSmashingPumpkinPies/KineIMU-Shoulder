"""Freeze CP3 delivery files; prove exact Git blob/disk bytes at delivery commit."""
import json
import subprocess
import sys
from hashlib import sha256
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
EVIDENCE = Path(__file__).resolve().parent
MAP = EVIDENCE / "SHA256SUMS.json"


def write(path, value):
    with path.open("x", encoding="utf-8", newline="\n") as stream:
        stream.write(json.dumps(value, sort_keys=True, indent=2) + "\n")


if sys.argv[1] == "snapshot":
    assert not MAP.exists()
    batch = EVIDENCE / "batch01"
    assert json.loads((batch / "summary.json").read_bytes())["cp3_passed"]
    assert json.loads((EVIDENCE / "retained-recompute.json").read_bytes())["summary"]["cp3_passed"]
    assert json.loads((EVIDENCE / "report-values-audit.json").read_bytes())["passed"]
    assert json.loads((EVIDENCE / "tamper-controls.json").read_bytes())["passed"]
    for name in ("full-final", "ruff-delivery", "mypy-delivery", "mypy-v11", "build-final",
                 "docs-final", "verify-report-final", "whitespace-final"):
        assert json.loads((EVIDENCE / f"{name}-check.json").read_bytes())["exit_code"] == 0, name
    full = (EVIDENCE / "full-final.log").read_text()
    assert "878 passed" in full and "skipped" not in full
    names = subprocess.check_output(["git", "ls-files", "--cached", "--others", "--exclude-standard", "--",
                                     EVIDENCE.relative_to(ROOT).as_posix(), "benchmarks/demo/run_benchmark.py",
                                     "benchmarks/demo/worker.py", "benchmarks/demo/PROTOCOL.md",
                                     "benchmarks/demo/REPORT.md", "benchmarks/demo/REPRODUCE.md",
                                     "tests/integration/test_m6_benchmark.py", "README.md"],
                                    cwd=ROOT, text=True).splitlines()
    names = sorted(set(names) - {MAP.relative_to(ROOT).as_posix()})
    files = {name: dict(bytes=len((ROOT / name).read_bytes()),
                        sha256=sha256((ROOT / name).read_bytes()).hexdigest()) for name in names}
    write(MAP, dict(source_commit=json.loads((batch / "lock.json").read_bytes())["source_head"], files=files,
                    scope="CP3 delivery snapshot; excludes dynamic state, self and later committed proof"))
    print("frozen delivery files", len(files))
elif sys.argv[1] == "prove":
    ref = sys.argv[2]
    map_name = MAP.relative_to(ROOT).as_posix()
    assert subprocess.check_output(["git", "show", f"{ref}:{map_name}"], cwd=ROOT) == MAP.read_bytes()
    frozen = json.loads(MAP.read_bytes())
    for name, expected in frozen["files"].items():
        disk = (ROOT / name).read_bytes()
        blob = subprocess.check_output(["git", "show", f"{ref}:{name}"], cwd=ROOT)
        assert disk == blob and dict(bytes=len(blob), sha256=sha256(blob).hexdigest()) == expected, name
    write(ROOT / sys.argv[3], dict(passed=True, delivery_commit=ref, source_commit=frozen["source_commit"],
                                   file_count=len(frozen["files"]),
                                   snapshot_map_sha256=sha256(MAP.read_bytes()).hexdigest(),
                                   git_disk_exact=True, files=frozen["files"]))
    print("committed bytes verified", ref, len(frozen["files"]))
else:
    raise ValueError("choose snapshot or prove")
