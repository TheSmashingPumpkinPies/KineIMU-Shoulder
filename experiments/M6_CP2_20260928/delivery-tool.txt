"""CP2 exact-byte snapshot/proof, adapted from the retained CP1 delivery helper."""
import json
import subprocess
import sys
from hashlib import sha256
from pathlib import Path

root = Path(__file__).resolve().parents[2]
evidence = Path(__file__).resolve().parent
map_path = evidence / "SHA256SUMS.json"
if sys.argv[1] == "snapshot":
    assert not map_path.exists()
    collection = json.loads((evidence / "collection.json").read_bytes())
    assert collection["complete"] and collection["inputs_unchanged"]
    assert json.loads((evidence / "pair-audit.json").read_bytes())["cp2_passed"]
    paths = [root / name for name in ("kineimu_shoulder/validation/demo.py", "examples/m6_demo.py",
                                     "tests/integration/test_m6_demo.py", "docs/M6_DEMO.md",
                                     ".github/workflows/python-ci.yml")]
    # Git's non-ignored inventory excludes runtime Python caches, never scientific products.
    names = subprocess.check_output(["git", "ls-files", "--cached", "--others", "--exclude-standard", "--",
                                     evidence.relative_to(root).as_posix()], cwd=root, text=True).splitlines()
    paths += [root / name for name in names if Path(name).name != "SHA256SUMS.json"]
    entries = {p.relative_to(root).as_posix(): dict(bytes=len(p.read_bytes()),
                                                 sha256=sha256(p.read_bytes()).hexdigest()) for p in sorted(paths)}
    report = dict(scope="CP2 wrapper/entry/test/guide/CI and evidence snapshot; excludes dynamic state, self, caches, later proof",
                  source_commit=collection["source_head"], files=entries)
    with map_path.open("x", encoding="utf-8", newline="\n") as stream:
        stream.write(json.dumps(report, indent=2) + "\n")
    print("snapshot", len(entries))
elif sys.argv[1] == "prove":
    ref = sys.argv[2]
    output = root / sys.argv[3]
    assert not output.exists()
    frozen = json.loads(map_path.read_bytes())["files"]
    map_name = map_path.relative_to(root).as_posix()
    assert subprocess.check_output(["git", "show", f"{ref}:{map_name}"], cwd=root) == map_path.read_bytes()
    rows = []
    for name, expected in frozen.items():
        disk = (root / name).read_bytes()
        blob = subprocess.check_output(["git", "show", f"{ref}:{name}"], cwd=root)
        assert disk == blob and dict(bytes=len(blob), sha256=sha256(blob).hexdigest()) == expected, name
        rows.append(dict(path=name, **expected, git_disk_exact=True))
    report = dict(delivery_commit=ref, source_commit=json.loads(map_path.read_bytes())["source_commit"],
                  passed=True, file_count=len(rows), files=rows,
                  snapshot_map_sha256=sha256(map_path.read_bytes()).hexdigest(),
                  scope="exact committed/disk bytes, no EOL equivalence; numerical acceptance in original pair audit")
    with output.open("x", encoding="utf-8", newline="\n") as stream:
        stream.write(json.dumps(report, indent=2) + "\n")
    print("committed proof", ref, len(rows))
else:
    raise ValueError("choose snapshot or prove")
