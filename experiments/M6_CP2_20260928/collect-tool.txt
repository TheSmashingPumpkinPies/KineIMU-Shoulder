"""Two fresh CLI processes; preserve original roots and exact-byte evidence copies."""
import json
import os
import shutil
import subprocess
import sys
import tempfile
import zipfile
from datetime import UTC, datetime
from hashlib import sha256
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
EVIDENCE = Path(__file__).resolve().parent
UV = Path(os.environ["USERPROFILE"]) / ".local/bin/uv.exe"


def snapshot(root):
    return {p.relative_to(root).as_posix(): sha256(p.read_bytes()).hexdigest()
            for p in sorted(root.rglob("*")) if p.is_file()}


def write(name, value):
    with (EVIDENCE / name).open("x", encoding="utf-8", newline="\n") as stream:
        stream.write(json.dumps(value, sort_keys=True, indent=2) + "\n")


assert not (EVIDENCE / "collection.json").exists()
head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
assert not subprocess.check_output(["git", "status", "--porcelain", "--untracked-files=no"], cwd=ROOT)
before = {"original": snapshot(ROOT / "experiments/M5_CP1_20260926/run1"),
          "sample": snapshot(ROOT / "datasets/samples/m6_synthetic")}
assert before == json.loads((EVIDENCE / "inputs-before.json").read_bytes())
parent = Path(tempfile.mkdtemp(prefix="kineimu-m6-cp2-formal-"))
environment = dict(os.environ)
environment.pop("KINEIMU_M1_RAW_ROOT", None)
for name in ("OPENBLAS_NUM_THREADS", "OMP_NUM_THREADS", "MKL_NUM_THREADS"):
    environment[name] = "1"
attempts = []
for index in (1, 2):
    output = parent / f"run{index}"
    command = [str(UV), "run", "--frozen", "python", "examples/m6_demo.py", "--output", str(output)]
    started = datetime.now(UTC).isoformat()
    process = subprocess.run(command, cwd=ROOT, env=environment,
                             stdout=subprocess.PIPE, stderr=subprocess.STDOUT, check=False)
    log = EVIDENCE / f"formal-run{index}.log"
    with log.open("xb") as stream:
        stream.write(process.stdout)
    row = dict(command=command, exit_code=process.returncode, started_utc=started,
               ended_utc=datetime.now(UTC).isoformat(), output_root=str(output), log=log.name,
               log_sha256=sha256(process.stdout).hexdigest(), product_sha256=snapshot(output),
               environment={key: environment.get(key) for key in
                            ("KINEIMU_M1_RAW_ROOT", "OPENBLAS_NUM_THREADS", "OMP_NUM_THREADS", "MKL_NUM_THREADS")})
    write(f"formal-run{index}-check.json", row)
    attempts.append(row)
    print(json.dumps(dict(attempt=index, exit_code=process.returncode, output=str(output))), flush=True)
    if process.returncode:
        break
after = {"original": snapshot(ROOT / "experiments/M5_CP1_20260926/run1"),
         "sample": snapshot(ROOT / "datasets/samples/m6_synthetic")}
write("inputs-after.json", after)
collection = dict(source_head=head, original_parent=str(parent), attempts=attempts,
                  inputs_unchanged=before == after, complete=len(attempts) == 2 and
                  all(row["exit_code"] == 0 for row in attempts))
write("collection.json", collection)
assert before == after and collection["complete"]
assert subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip() == head
assert not subprocess.check_output(["git", "status", "--porcelain", "--untracked-files=no"], cwd=ROOT)
command = [sys.executable, str(EVIDENCE / "audit.py"), "--run1", str(parent / "run1"),
           "--run2", str(parent / "run2"), "--output", str(EVIDENCE / "pair-audit.json")]
process = subprocess.run(command, cwd=ROOT, env=environment, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
with (EVIDENCE / "pair-audit.log").open("xb") as stream:
    stream.write(process.stdout)
write("pair-audit-check.json", dict(command=command, exit_code=process.returncode,
                                    log="pair-audit.log", sha256=sha256(process.stdout).hexdigest()))
assert process.returncode == 0, process.stdout.decode(errors="replace")
retained = {}
for index in (1, 2):
    source = parent / f"run{index}"
    target = EVIDENCE / f"run{index}"
    assert not target.exists()
    shutil.copytree(source, target)
    assert snapshot(source) == snapshot(target) == attempts[index - 1]["product_sha256"]
    retained[f"run{index}"] = snapshot(target)
write("retained-products.json", retained)
# Freeze the actual executed repository bytes, independent of Git line-ending conversion.
run = json.loads((parent / "run1/run.json").read_bytes())
manifest = json.loads((parent / "run1/replay/manifest.json").read_bytes())
sources = {**manifest["source_file_sha256"], **run["source"]["file_sha256"]}
for name in ("experiments/M6_CP2_20260928/audit.py", "experiments/M6_CP2_20260928/collect-tool.txt",
             "experiments/M6_CP2_20260928/check-tool.txt", "docs/M6_DEMO.md",
             "tests/integration/test_m6_demo.py", ".github/workflows/python-ci.yml"):
    sources[name] = sha256((ROOT / name).read_bytes()).hexdigest()
with zipfile.ZipFile(EVIDENCE / "runtime-source.zip", "x", compression=zipfile.ZIP_DEFLATED) as archive:
    for name, digest in sorted(sources.items()):
        content = (ROOT / name).read_bytes()
        assert sha256(content).hexdigest() == digest
        archive.writestr(name, content)
with zipfile.ZipFile(EVIDENCE / "runtime-source.zip") as archive:
    assert {name: sha256(archive.read(name)).hexdigest() for name in archive.namelist()} == sources
write("runtime-source-sha256.json", sources)
print(process.stdout.decode(errors="replace"), flush=True)
print(json.dumps(dict(cp2_pair_passed=True, retained_files_per_run=26, source_files=len(sources))), flush=True)
